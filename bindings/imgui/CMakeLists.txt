# bui_imgui - Dear ImGui (vendored, extern/imgui) running on BUI: a platform +
# renderer backend (imgui_impl_bui.cpp) and Blender's look (imgui_bui_style.cpp).
# See README.md.
#
# SPDX-License-Identifier: GPL-2.0-or-later

set(IMGUI_DIR ${CMAKE_CURRENT_SOURCE_DIR}/extern/imgui)
# Blender widgets: ImGui's widgets drawn and handled as Blender's (blender/README.md). The
# copy blender/imgui_widgets_blender.cpp replaces imgui_widgets.cpp: same API and ABI, so
# the C / Python / Rust bindings get Blender widgets unchanged.
option(BUI_IMGUI_BLENDER_WIDGETS "Dear ImGui on BUI: Blender's widgets (blender/imgui_widgets_blender.cpp)" OFF)
if(BUI_IMGUI_BLENDER_WIDGETS)
  set(BUI_IMGUI_WIDGETS_SRC ${CMAKE_CURRENT_SOURCE_DIR}/blender/imgui_widgets_blender.cpp)
else()
  set(BUI_IMGUI_WIDGETS_SRC ${IMGUI_DIR}/imgui_widgets.cpp)
endif()
set(BUI_IMGUI_CORE
  ${IMGUI_DIR}/imgui.cpp
  ${IMGUI_DIR}/imgui_draw.cpp
  ${IMGUI_DIR}/imgui_tables.cpp
  ${BUI_IMGUI_WIDGETS_SRC}
  ${IMGUI_DIR}/imgui_demo.cpp
  ${IMGUI_DIR}/misc/cpp/imgui_stdlib.cpp
)
set(BUI_IMGUI_BACKEND
  ${CMAKE_CURRENT_SOURCE_DIR}/imgui_impl_bui.cpp
  ${CMAKE_CURRENT_SOURCE_DIR}/imgui_bui_style.cpp
)
# Dear ImGui's own sources keep their upstream warning level.
set_source_files_properties(${BUI_IMGUI_CORE} PROPERTIES COMPILE_OPTIONS "$<IF:$<CXX_COMPILER_ID:MSVC>,/W1,-w>")

add_library(bui_imgui STATIC ${BUI_IMGUI_CORE} ${BUI_IMGUI_BACKEND})
add_library(bui::imgui ALIAS bui_imgui)
target_include_directories(bui_imgui PUBLIC
  ${CMAKE_CURRENT_SOURCE_DIR}
  ${IMGUI_DIR}
  ${IMGUI_DIR}/misc/cpp
)
target_link_libraries(bui_imgui PUBLIC bui::kit)
if(BUI_IMGUI_BLENDER_WIDGETS)
  target_compile_definitions(bui_imgui PUBLIC BUI_IMGUI_BLENDER_WIDGETS=1)
endif()

# The C API (cimgui-style, generated by gen/gen.py) and the C API of the BUI backend.
add_library(bui_cimgui STATIC c/cimgui.cpp c/cimgui_impl_bui.cpp)
add_library(bui::cimgui ALIAS bui_cimgui)
target_include_directories(bui_cimgui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/c)
target_include_directories(bui_cimgui PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
target_link_libraries(bui_cimgui PUBLIC bui_imgui)
if(MSVC)
  # offsetof on imgui's C++ structs (layout checks), large generated file
  target_compile_options(bui_cimgui PRIVATE /bigobj /wd4190)
endif()

# examples/imgui_demo.cpp: ImGui::ShowDemoWindow() in a BUI window (or headless with
# --screenshot); examples/imgui_demo_c.c: the same from C. (Not in source copies
# without examples/, e.g. the Rust crate's vendored csrc/.)
if(EXISTS ${BUI_SOURCE_DIR}/examples/imgui_demo.cpp)
  add_executable(bui_imgui_demo ${BUI_SOURCE_DIR}/examples/imgui_demo.cpp)
  target_include_directories(bui_imgui_demo PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  target_link_libraries(bui_imgui_demo PRIVATE bui::imgui)
endif()
# examples/imgui_blender_compare.cpp: ImGui's widgets next to the kit's native ones (PNGs)
if(EXISTS ${BUI_SOURCE_DIR}/examples/imgui_blender_compare.cpp)
  add_executable(bui_imgui_compare ${BUI_SOURCE_DIR}/examples/imgui_blender_compare.cpp)
  target_include_directories(bui_imgui_compare PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  target_link_libraries(bui_imgui_compare PRIVATE bui::imgui)
  if(BUI_BUILD_TESTS AND BUI_IMGUI_BLENDER_WIDGETS)
    # ImGui's Blender widgets and the kit's native widgets: the same pixels
    add_test(NAME imgui_blender_widgets_match_kit COMMAND bui_imgui_compare ${CMAKE_CURRENT_BINARY_DIR})
    set_tests_properties(imgui_blender_widgets_match_kit PROPERTIES TIMEOUT 300)
  endif()
endif()
if(EXISTS ${BUI_SOURCE_DIR}/examples/imgui_demo_c.c)
  add_executable(bui_imgui_demo_c ${BUI_SOURCE_DIR}/examples/imgui_demo_c.c)
  target_link_libraries(bui_imgui_demo_c PRIVATE bui::cimgui)
endif()

if(BUI_BUILD_TESTS)
  # The test compiles its own Dear ImGui with the test-engine hooks (to open every
  # section of the demo), so it does not link bui_imgui.
  add_executable(bui_imgui_test tests/test_imgui_bui.cpp ${BUI_IMGUI_CORE} ${BUI_IMGUI_BACKEND})
  target_compile_definitions(bui_imgui_test PRIVATE IMGUI_ENABLE_TEST_ENGINE)
  if(BUI_IMGUI_BLENDER_WIDGETS)
    target_compile_definitions(bui_imgui_test PRIVATE BUI_IMGUI_BLENDER_WIDGETS=1)
  endif()
  target_include_directories(bui_imgui_test PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR} ${IMGUI_DIR} ${IMGUI_DIR}/misc/cpp ${CMAKE_CURRENT_SOURCE_DIR}/tests)
  target_link_libraries(bui_imgui_test PRIVATE bui::kit)
  add_test(NAME imgui_backend COMMAND bui_imgui_test)
  set_tests_properties(imgui_backend PROPERTIES TIMEOUT 300)

  # the C API, headless, from C
  add_executable(bui_cimgui_test tests/test_cimgui.c)
  target_link_libraries(bui_cimgui_test PRIVATE bui::cimgui)
  add_test(NAME imgui_c_api COMMAND bui_cimgui_test)
  set_tests_properties(imgui_c_api PROPERTIES TIMEOUT 300)
endif()
